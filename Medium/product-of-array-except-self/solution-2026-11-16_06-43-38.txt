class Solution {
public:
    vector<int> productExceptSelf(vector<int>& nums) {

        std::vector<int> pre(nums.size(), 1);
        for (int i = 1; i < nums.size(); i++) {
            pre[i] = pre[i-1] * nums[i-1];
        }

        int post = 1;
        for (int j = nums.size() - 1; j >= 0; j--) {
            int temp = nums[j];
            pre[j] = post * pre[j];
            post *= temp;
        }

        return pre;
    }
};
