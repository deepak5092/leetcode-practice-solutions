class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        std::unordered_set<char> visit;
        int l = 0;
        int res = 0;
        char curr {};
        char lChar {};

        for (int r = 0; r < s.size(); r++) {
            curr = s[r];
            if (!visit.contains(curr)) {
                visit.insert(curr);
                res = max(res, r - l + 1);
                continue;
            }

            while (visit.contains(curr)) {
                lChar = s[l];
                visit.erase(lChar);
                l++;
            }

            visit.insert(curr);
            res = max(res, r - l + 1);
        }
        return res;
    }
};
