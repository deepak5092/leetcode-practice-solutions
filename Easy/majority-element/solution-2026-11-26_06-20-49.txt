class Solution {
public:
    int majorityElement(vector<int>& nums) {
        
        double maj {static_cast<double>(nums.size())/2};
        std::unordered_map<int, int> counter;

        for (int n : nums) {
            counter[n]++;
        }

        for (const auto& [num, cntr] : counter) {
            if (cntr > maj) {
                return num;
            }
        }

        return 0;
    }
};
