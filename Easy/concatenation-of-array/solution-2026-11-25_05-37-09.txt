class Solution {
public:
    vector<int> getConcatenation(vector<int>& nums) {
        
        int len = nums.size();
        std::vector<int> res(2 * len);

        for (int i = 0; i < len; i++) {
            res[i] = nums[i];
            res[i+len] = nums[i];
        }

        return res;
    }
};
