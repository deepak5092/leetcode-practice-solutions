#include <unordered_map>

class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {
        
        std::unordered_map<int, int> idxMap;

        for (int i = 0; i < nums.size(); i++) {
            idxMap[nums[i]] = i;
        }

        for (int j = 0; j < nums.size(); j++) {
            int curr = nums[j];
            int lookup = target - curr;

            if (idxMap.find(lookup) != idxMap.end() && idxMap[lookup] != j) {
                return {idxMap[lookup], j};
            }
        }

        return {0};
    }
};
